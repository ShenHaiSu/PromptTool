### 4.2 关键实现

**快捷键**

```ts
// src/composables/useShortcuts.ts
import { useMagicKeys } from '@vueuse/core'
export function useShortcuts(handlers: { focusSearch: () => void; save: () => void; copy: () => void }) {
  const { Ctrl_F, Ctrl_S, Ctrl_C } = useMagicKeys()
  whenever(Ctrl_F, handlers.focusSearch)
  whenever(Ctrl_S, handlers.save)
  whenever(Ctrl_C, handlers.copy)
}
```

**CSV 导出（前端版，无需 Rust）**

