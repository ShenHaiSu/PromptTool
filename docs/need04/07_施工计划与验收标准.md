## 二、新增单测（3 例为门槛）

```ts
// assembly.test.ts
it('dimOrderMap 插队：新建维度 sortOrder=0 排到 gender 前', () => {
  const items = [top, newDim]; // newDim.dimensionKey='my_style'
  expect(sortByOrder(items, 'dimensionOrder', { top: 6, my_style: 0 })[0].module.dimensionKey).toBe('my_style')
})
it('缺键沉底不崩', () => {
  expect(() => sortByOrder(items, 'dimensionOrder', {})).not.toThrow()
})
// random.test.ts
it('randomAssembly 尊重 sortOrder：新维度在前', () => {
  const dimensions = [dim('my_style', 0), dim('top', 6), dim('bottom', 7)]
  const [ir] = randomAssembly(dimensions, modulesByDim, new Set(), 1, defaultConfig())
  expect(ir.segments[0].dimensionKey).toBe('my_style')
})
```
