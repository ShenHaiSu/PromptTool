function onExport(): void {
  if (!props.prompt?.trim()) { push('暂无可导出的 Prompt','warning'); return }
  if (!props.ir) {

    exportSingleCsv(fakeIr, props.prompt); push('已导出 CSV','success',1500); return
  }

}

### 5.2 新增：浅底色与辅助

```ts
import { dimColor } from '@/lib/utils'

function selectedCardBg(key: string): string {
  const hex = dimColor(key ?? '')
  // light 08 / dark 16 — 内联 rgba，避免新增 CSS 变量
  const m = hex.match(/^#([0-9a-f]{6})$/i)
  if (!m) return ''
  const n = parseInt(m[1]!, 16)
  const r = (n >> 16) & 0xff, g = (n >> 8) & 0xff, b = n & 0xff
  const isDark = document.documentElement.classList.contains('dark')
  const a = isDark ? 0.16 : 0.08
  return `rgba(${r},${g},${b},${a})`
}
```

- 深色模式通过 `documentElement.classList.contains('dark')` 判断（与 `useThemeStore` 同源）；
- 可选：监听 `themeStore.mode` 触发布局重算，若卡片已内联 `rgba` 则无需响应式（下次渲染即更新）。

## 7. 关键边界与校验

| 校验点 | 规则 |
|--------|------|
| 空 `dimensionKey` | `dimColor('') → #94A3B8`，浅底色回落灰色，标签显示 `未分类` |
| `keyword` 在已选模式 | 仅过滤已选卡片，不影响 `assembly`；清空按钮重置 `keyword=''` |
| `allowNsfw` 在已选模式 | 不生效，已选项恒展示（含 NSFW） |
| 权重越界 | `clampWeight 0.5~2.0 step 0.1`，确认后 `v===1.0 ? null : v` 写回，避免 `1.0` 的无意义 override |
| 拖拽在 `dimensionOrder` | 保留手柄，`onDragEnd` 仍同步 `selectedItems` 但 `sortBy` 为维度顺序时 IR 顺序不受展示顺序影响；可 toast 提示 |
| 预览空态 | `isEmpty` 时 `preview-empty` + 复制/导出 `disabled` |
| 深色模式 | `selectedCardBg` 的 `a=0.16` 保证可辨；`bg-muted/30` 与 `bg-popover` 保持现有 token |
| 长词条名 | `truncate max-w-[10rem]` 或 `min-w-0 flex-1`，`title=contentEn` 兜底 |
| 并发保存 | `SaveDialog` 的 `confirm` 为 async，按钮 `disabled` 由调用方控制（可选） |
